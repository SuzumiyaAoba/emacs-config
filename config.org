* init.el

=packlet= はビルド時だけ全体を読み込んでマクロを展開し、起動時は小さな runtime だけを
読み込む。=emacs-batteries= も Nix で byte-compile 済みのものを使う。

#+begin_src emacs-lisp :tangle init.el
;; init.el --- Minimal emacs-batteries setup -*- lexical-binding: t; coding: utf-8 -*-

(defconst my/config-root
  (file-name-as-directory
   (expand-file-name
    (or (getenv "EMACS_CONFIG_ROOT")
        (file-name-directory (or load-file-name user-init-file))))))

(eval-when-compile
  (require 'packlet)
  ;; Expand the deferred hydra definition while compiling init.el.
  (require 'pretty-hydra))

(require 'packlet-runtime)
(autoload 'packlet "packlet" nil nil 'macro)

;; Compatibility for the pinned packlet revision.  Newer packlet versions
;; provide this runtime helper without loading the parser.
(unless (fboundp 'packlet--idle-delay)
  (defun packlet--idle-delay (value)
    "Normalize runtime VALUE for packlet's `:idle' keyword."
    (cond
     ((null value) nil)
     ((eq value t) 1.0)
     ((and (numberp value) (>= value 0)) value)
     (t
      (error
       "packlet: :idle must evaluate to t, nil, or a non-negative number")))))

(require 'emacs-batteries)
(emacs-batteries-setup)

;; Build-time expansion sites have no source scope, so their cleanup objects
;; can never be used.  Avoid allocating that bookkeeping while retaining full
;; behavior when source tracking is loaded for interactive evaluation later.
(defun my/packlet-register-static-entry (original scope id install cleanup)
  "Install a compiled packlet entry without unused source bookkeeping."
  (if (or scope packlet--active-source-scope)
      (funcall original scope id install cleanup)
    (funcall install)))
(advice-add 'packlet--register-source-entry
            :around #'my/packlet-register-static-entry)

;; Nix already exposes every package on `load-path'.  Avoid searching for and
;; loading dozens of separate *-autoloads files; packlet installs a direct
;; fallback autoload for each configured command, mode, and hook instead.
(defun my/packlet-skip-package-autoloads (&rest _args)
  "Let packlet install fallback autoloads without loading package autoload files."
  nil)
(advice-add 'packlet--maybe-load-autoloads
            :override #'my/packlet-skip-package-autoloads)

;; Persistent history, auto-revert, which-key, and editorconfig are useful
;; after the first frame but do not need to delay it.
(defun my/schedule-emacs-batteries-services ()
  "Start deferred emacs-batteries services after Emacs becomes idle."
  (remove-hook 'emacs-startup-hook
               #'my/schedule-emacs-batteries-services)
  (run-with-idle-timer 1.0 nil #'emacs-batteries--run-deferred))
(when emacs-batteries--deferred-functions
  (remove-hook 'emacs-startup-hook #'emacs-batteries--run-deferred)
  (add-hook 'emacs-startup-hook
            #'my/schedule-emacs-batteries-services 90))
#+end_src

* Input method

=DDSKK= を日本語入力として使う。

#+begin_src emacs-lisp :tangle init.el
(packlet skk
  :setq ((default-input-method "japanese-skk")
         (skk-user-directory (expand-file-name "skk/" user-emacs-directory)))
  :autoload ((skk-activate "skk-leim" t)
             (skk-auto-fill-activate "skk-leim" t)
             (skk-isearch-mode-setup "skk-isearch")
             (skk-isearch-mode-cleanup "skk-isearch"))
  :hook-if-feature ((isearch-mode-hook skk skk-isearch-mode-setup)
                    (isearch-mode-end-hook skk skk-isearch-mode-cleanup))
  :init
  ;; DDSKK 本体の autoload を読み込む。これがないと `skk-mode-invoke' が呼ぶ
  ;; `skk-cus-setup' などが未定義のままで void-function エラーになる。
  (require 'ddskk-autoloads nil t)
  (require 'ccc-autoloads nil t)
  (require 'ddskk-posframe-autoloads nil t)
  (make-directory skk-user-directory t)
  (register-input-method
   "japanese-skk" "Japanese" #'skk-activate ""
   "Simple Kana to Kanji conversion program")
  (register-input-method
   "japanese-skk-auto-fill" "Japanese" #'skk-auto-fill-activate ""
   "Simple Kana to Kanji conversion program with auto-fill")
  (defun my/skk-handle-c-j ()
    "Enable SKK or return to hiragana mode on `C-j'."
    (interactive)
    (cond
     ((not (bound-and-true-p skk-mode))
      (let ((input-method-verbose-flag nil))
        (activate-input-method default-input-method)))
     ((bound-and-true-p skk-henkan-mode)
      (call-interactively #'skk-kakutei))
     ((or (bound-and-true-p skk-latin-mode)
          (bound-and-true-p skk-jisx0208-latin-mode)
          (bound-and-true-p skk-abbrev-mode)
          (bound-and-true-p skk-katakana))
      (skk-j-mode-on))
     (t
      (call-interactively #'skk-kakutei))))
  (defvar my/skk-shortcut-mode-map
    (let ((map (make-sparse-keymap)))
      (define-key map (kbd "C-j") #'my/skk-handle-c-j)
      (define-key map (kbd "C-x j") #'toggle-input-method)
      map))
  (define-minor-mode my/skk-shortcut-mode
    "Provide global keybindings for SKK."
    :global t
    :group 'convenience
    :lighter ""
    :keymap my/skk-shortcut-mode-map)
  (defun my/skk-auto-enable-p ()
    "Return non-nil when SKK should be enabled automatically."
    (and (not noninteractive)
         (not (minibufferp))
         (not buffer-read-only)
         (not (derived-mode-p 'special-mode))))
  (defun my/enable-skk-by-default ()
    "Enable SKK by default in editable interactive buffers.
When SKK is newly enabled, switch to ASCII mode."
    (cond
     ((my/skk-auto-enable-p)
      (when (null current-input-method)
        (let ((input-method-verbose-flag nil))
          (activate-input-method default-input-method))
        (when (and (bound-and-true-p skk-mode)
                   (fboundp 'skk-latin-mode))
          (skk-latin-mode 1))))
     ((and (stringp current-input-method)
           (string-prefix-p "japanese-skk" current-input-method))
      (let ((input-method-verbose-flag nil))
        (deactivate-input-method)))))
  (my/skk-shortcut-mode 1)
  :hook (after-init-hook my/enable-skk-by-default 1.0)
  :commands (skk-mode))

(packlet ddskk-posframe
  :after skk
  :when (display-graphic-p)
  :enable (ddskk-posframe-mode t))
#+end_src

* Startup

ファイル指定なしで起動したときは、最初のフレームを表示した後の idle 時に
起動ディレクトリを =dired= で開く。先に入力が始まった場合はバッファを奪わない。

#+begin_src emacs-lisp :tangle init.el
(packlet startup
  :init
  (defun my/open-startup-directory-in-dired (buffer directory)
    "Open DIRECTORY in Dired when BUFFER is still the untouched *scratch*."
    (when (and (buffer-live-p buffer)
               (eq buffer (window-buffer (selected-window)))
               (file-directory-p directory)
               (with-current-buffer buffer
                 (and (string= (buffer-name) "*scratch*")
                      (not (buffer-modified-p)))))
      (switch-to-buffer (dired-noselect directory))))
  (defun my/schedule-startup-directory ()
    "Schedule the initial Dired buffer after the first frame is usable."
    (when-let ((buffer (get-buffer "*scratch*")))
      (run-with-idle-timer
       0.1 nil
       #'my/open-startup-directory-in-dired
       buffer
       (buffer-local-value 'default-directory buffer))))
  :hook (window-setup-hook . my/schedule-startup-directory))
#+end_src

* Project

=projectile= を有効にし、lazy prefix key で project 操作をまとめる。

#+begin_src emacs-lisp :tangle init.el
(defun my/project-dispatch (consult-fn projectile-fn &optional arg)
  "Call CONSULT-FN if consult-projectile is available and ARG is nil.
Otherwise call PROJECTILE-FN, passing ARG when non-nil."
  (require 'projectile)
  (if (and (not arg) (require 'consult-projectile nil t))
      (funcall consult-fn)
    (if arg (funcall projectile-fn arg) (funcall projectile-fn))))

(defun my/project-switch-to-buffer ()
  "Switch to a project buffer, preferring consult-projectile."
  (interactive)
  (my/project-dispatch #'consult-projectile-switch-to-buffer
                       #'projectile-switch-to-buffer))

(defun my/project-find-dir (&optional arg)
  "Find a project directory, preferring consult-projectile.
With prefix ARG, bypass consult-projectile and invalidate the cache."
  (interactive "P")
  (my/project-dispatch #'consult-projectile-find-dir
                       #'projectile-find-dir arg))

(defun my/project-find-file (&optional arg)
  "Find a project file, preferring consult-projectile.
With prefix ARG, bypass consult-projectile and invalidate the cache."
  (interactive "P")
  (my/project-dispatch #'consult-projectile-find-file
                       #'projectile-find-file arg))

(defun my/project-recentf ()
  "Browse recent project files, preferring consult-projectile."
  (interactive)
  (my/project-dispatch #'consult-projectile-recentf
                       #'projectile-recentf))

(defun my/project-switch-project (&optional arg)
  "Switch projects, preferring consult-projectile.
With prefix ARG, bypass consult-projectile."
  (interactive "P")
  (my/project-dispatch #'consult-projectile-switch-project
                       #'projectile-switch-project arg))

(packlet projectile
  :setq ((projectile-cache-file
          (expand-file-name "projectile-cache.eld" user-emacs-directory))
         (projectile-known-projects-file
          (expand-file-name "projectile-bookmarks.eld" user-emacs-directory)))
  :idle 1.0
  :bind-keymap ("C-c p" . projectile-command-map)
  :bind-after-load
  (projectile
   (:map projectile-command-map
    ("b" . my/project-switch-to-buffer)
    ("d" . my/project-find-dir)
    ("e" . my/project-recentf)
    ("f" . my/project-find-file)
    ("p" . my/project-switch-project)
    ("s r" . my/consult-ripgrep)))
  :after-load
  (projectile
   (setq consult-project-function
         (lambda (&optional may-prompt)
           (or (projectile-project-root)
               (when may-prompt
                 (projectile-switch-project-by-name
                  (projectile-completing-read
                   "Switch to project: "
                   (projectile-relevant-known-projects)))
                 (projectile-project-root))))))
  :enable projectile-mode)

(packlet consult-projectile
  :after (consult projectile))
#+end_src
